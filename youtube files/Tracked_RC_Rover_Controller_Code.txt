void setup() {
  Serial.begin(115200);
  analogReadResolution(12);

  // Buton ve switchler için dâhili pull-up dirençlerini aktifleÅŸtiriyoruz
  pinMode(switch1, INPUT_PULLUP);
  pinMode(switch2, INPUT_PULLUP);
  pinMode(button1, INPUT_PULLUP);
  pinMode(button2, INPUT_PULLUP);

  // --- nRF24L01 Kurulumu ---
  // ESP32-C6 için varsayÄ±lan SPI pinleri kullanÄ±lÄ±r: MOSI=19, MISO=20, SCK=21
  if (!radio.begin()) {
    Serial.println("DIKKAT: nRF24L01 Modulu Bulunamadi!");
    Serial.println("Web arayuzu calismaya devam edecek ancak RF ile veri gonderilmeyecek.");
    isRadioConnected = false;
  } else {
    isRadioConnected = true;
    radio.openWritingPipe(radioPipe);
    radio.setPALevel(RF24_PA_MAX); // DP5 modülü için güç ayarÄ±. Sorun olursa RF24_PA_LOW yap.
    radio.setDataRate(RF24_250KBPS);
    radio.stopListening();
    Serial.println("nRF24L01 Basariyla Baslatildi (Verici Modu).");
  }

  // --- Wi-Fi ve Captive Portal Kurulumu ---
  WiFi.mode(WIFI_AP);
  WiFi.softAP(ssid);
  Serial.print("AP Baslatildi. Baglanti IP: "); 
  Serial.println(WiFi.softAPIP());
  
  dnsServer.start(DNS_PORT, "*", WiFi.softAPIP());

  server.on("/", []() { 
    server.send(200, "text/html", index_html); 
  });

  server.on("/veri", []() {
    String json = "{";
    json += "\"j1x\":" + String(sentData.j1x) + ",";
    json += "\"j1y\":" + String(sentData.j1y) + ",";
    json += "\"j2x\":" + String(sentData.j2x) + ",";
    json += "\"j2y\":" + String(sentData.j2y) + ",";
    json += "\"p1\":"  + String(sentData.p1)  + ",";
    json += "\"p2\":"  + String(sentData.p2)  + ",";
    json += "\"sw1\":" + String(sentData.sw1) + ",";
    json += "\"sw2\":" + String(sentData.sw2) + ",";
    json += "\"btn1\":"+ String(sentData.btn1)+ ",";
    json += "\"btn2\":"+ String(sentData.btn2);
    json += "}";
    server.send(200, "application/json", json);
  });

  server.onNotFound([]() { 
    server.send(200, "text/html", index_html); 
  });
  
  server.begin();
}

void loop() {
  dnsServer.processNextRequest();
  server.handleClient();

  // Her 50 milisaniyede bir donanÄ±mlarÄ± oku ve gönder
  if (millis() - lastSendTime >= 50) {
    lastSendTime = millis();

    // Verileri paket içine kaydet
    sentData.j1x = analogRead(joy1_X);
    sentData.j1y = analogRead(joy1_Y);
    sentData.j2x = analogRead(joy2_X);
    sentData.j2y = analogRead(joy2_Y);
    sentData.p1  = analogRead(pot1);
    sentData.p2  = analogRead(pot2);
    
    // INPUT_PULLUP olduÄŸu için LOW (0) = BasÄ±lÄ±. Arayüz için Tersliyoruz (!) 
    sentData.sw1  = !digitalRead(switch1);
    sentData.sw2  = !digitalRead(switch2);
    sentData.btn1 = !digitalRead(button1);
    sentData.btn2 = !digitalRead(button2);

    // EÄžER nRF24L01 sisteme düzgün baÄŸlanmÄ±ÅŸsa veriyi radyo frekansÄ±yla gönder
    if (isRadioConnected) {
      radio.write(&sentData, sizeof(Data_Package));
    }

    // --- Seri Monitöre YazdÄ±rma Bölümü ---
    Serial.print("J1 [X:"); Serial.print(sentData.j1x);
    Serial.print(" Y:"); Serial.print(sentData.j1y);
    Serial.print("] | J2 [X:"); Serial.print(sentData.j2x);
    Serial.print(" Y:"); Serial.print(sentData.j2y);
    Serial.print("] | Potlar [P1:"); Serial.print(sentData.p1);
    Serial.print(" P2:"); Serial.print(sentData.p2);
    Serial.print("] | SW [1:"); Serial.print(sentData.sw1);
    Serial.print(" 2:"); Serial.print(sentData.sw2);
    Serial.print("] | BTN [1:"); Serial.print(sentData.btn1);
    Serial.print(" 2:"); Serial.print(sentData.btn2);
    Serial.println("]");
  }
}
